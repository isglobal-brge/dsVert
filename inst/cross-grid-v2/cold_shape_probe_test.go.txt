package main

import (
 "bytes"
 "encoding/base64"
 "os"
 "testing"
 "time"
)

func TestCrossGridColdPoissonK3ShapeProbe(t *testing.T) {
 p := crossGridKernelCostPlan(t, "poisson")
 p.Owners = 3
 p.MaxOutcome = 16
 for j := range p.Caps { p.Caps[j] = 6205620 }
 dir := t.TempDir()
 if err := os.Chmod(dir, 0700); err != nil { t.Fatal(err) }
 cache := &crossGridCircuitCache{Directory: dir, Key: base64.StdEncoding.EncodeToString(bytes.Repeat([]byte{37},32))}
 before := time.Now()
 cold, err := crossGridKernelPrepareCached(p, cache)
 if err != nil { t.Fatal(err) }
 coldSeconds := time.Since(before).Seconds()
 before = time.Now()
 hot, err := crossGridKernelPrepareCached(p, cache)
 if err != nil { t.Fatal(err) }
 var a, b bytes.Buffer
 if cold.circuit.Marshal(&a) != nil || hot.circuit.Marshal(&b) != nil || !bytes.Equal(a.Bytes(),b.Bytes()) { t.Fatal("public topology mismatch") }
 t.Logf("public_shape rows=32 p=10 grid=8 owners=3 a=4 max_outcome=16 g=16 cold_seconds=%.6f hot_seconds=%.6f circuit_bytes=%d",coldSeconds,time.Since(before).Seconds(),a.Len())
 if coldSeconds >= 120 { t.Fatal("public cold compilation exceeds existing readiness window") }
}
